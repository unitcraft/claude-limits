// w3: two lambdas, each under `with` of a handler VALUE, capture one RECORD parameter.
module hcap.w3_record_param

type Where value {
    ro dir str
    ro file str
}

type Ask effect {
    ask() -> int
}

fn forty() -> Effect[Ask] => effect Ask { ask() -> int { 40 } }

fn run_ask(f fn() Ask -> int) -> int => f()

fn both(at Where, flag bool) -> int {
    ro n = 2
    ro a = with Ask = forty() { run_ask(fn() Ask -> int => Ask.ask() + at.dir.byte_len() + n + (if flag { 1 } else { 0 })) }
    ro m = 3
    ro b = with Ask = forty() { run_ask(fn() Ask -> int => Ask.ask() + at.file.byte_len() + m + (if flag { 1 } else { 0 })) }
    a + b
}

fn main() Io {
    ro w = Where{ dir: "ab", file: "cde" }
    println("w3 ${both(w, true)}")
}
