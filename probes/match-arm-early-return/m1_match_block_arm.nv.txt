// m1: `Ok(match ...)` whose first arm is a block with an early `return Err(...)` and a
// value at its end -- nova-duckdb `QueryResult @value`.
module hcap.m1_match_block_arm

type Val enum
    | Flag(bool)
    | Num(int)

fn read(k int) -> Result[Val, str] {
    Ok(match k {
        0 => {
            mut v = 1
            v = v + 1
            if v < 0 { return Err("negative") }
            Val.Flag(v != 0)
        }
        _ => Val.Num(k)
    })
}

fn show(v Val) -> str =>
    match v {
        Val.Flag(b) => "flag ${b}"
        Val.Num(n) => "num ${n}"
    }

fn main() Io {
    match read(0) {
        Ok(v) => println("m1 ${show(v)}")
        Err(e) => println("m1 err ${e}")
    }
}
