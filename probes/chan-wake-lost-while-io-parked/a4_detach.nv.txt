// a4: as a2, but each "connection" is a DETACHED fiber, and inside it the asker runs
// in a nested `supervised { spawn }` -- the shape polaris gives a connection
// (`serve`: `detach consume stream { serve_connection(..) }`, and the handler in
// `supervised { spawn consume shs = s.share() { handler(raw) } }`). No sockets.
module p1406.a4_detach

import std.concurrency.nvchan.{Chan, ChanTx, ChanRx}
import std.time.duration

type Ask value {
    ro n int
    ro reply ChanTx[int]
}

fn store(rx ChanRx[Ask]) -> () {
    mut going = true
    while going {
        match rx.recv() {
            None => { going = false }
            Some(a) => { ro _ = a.reply.send(a.n * 10) }
        }
    }
}

fn ask(tx ChanTx[Ask], n int) -> int {
    ro pair = Chan[int].new(1)
    ro _ = tx.send({ n, reply: pair.tx })
    pair.rx.recv() ?? (0 - 1)
}

fn handle(tx ChanTx[Ask], i int) Detach -> () {
    supervised {
        spawn {
            println("a4 conn ${i}: asking")
            ro got = ask(tx, i + 1)
            println("a4 conn ${i}: store said ${got}")
        }
    }
}

fn main() Io Time Detach {
    ro req = Chan[Ask].new(8)
    ro tx = req.tx
    supervised {
        spawn {
            for i in 0..3 {
                detach { handle(tx, i) }
            }
            // The detached fibers are nobody's children: give them time, then end the store.
            500.to_millis().sleep()
            println("a4 closing the store")
            tx.close()
        }
        store(req.rx)
    }
    println("a4 done")
}
