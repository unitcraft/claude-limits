// c4: c3 with a line printed when the STORE receives -- which side sleeps. c3: c1, plus a client that connects to the parked listener 2 s after the start.
// If the stuck wake is released by the next I/O event, the reply arrives ~2 s late.
module p1406.c4_who_sleeps

import std.concurrency.nvchan.{Chan, ChanTx, ChanRx}
import std.net.{TcpListener, TcpStream}
import std.os.{real_os, exit_process}
import std.time.duration

type Ask value {
    ro n int
    ro reply ChanTx[int]
}

fn store(rx ChanRx[Ask]) -> () {
    mut going = true
    while going {
        match rx.recv() {
            None => { going = false }
            Some(a) => { println("c4 store got the ask"); ro _ = a.reply.send(a.n * 10) }
        }
    }
}

fn ask(tx ChanTx[Ask], n int) -> int {
    ro pair = Chan[int].new(1)
    ro _ = tx.send({ n, reply: pair.tx })
    pair.rx.recv() ?? (0 - 1)
}

fn main() Io Net Time Detach {
    detach {
        5000.to_millis().sleep()
        println("c4 WATCHDOG: still running after 5 s -- hung")
        with Os = real_os() { exit_process(3) }
    }
    ro req = Chan[Ask].new(8)
    ro tx = req.tx
    consume l = match TcpListener.bind("127.0.0.1:0") {
        Ok(consume x) => x
        Err(e) => panic("bind: ${e}")
    }
    ro port = l.local_port()
    detach {
        2000.to_millis().sleep()
        println("c4 client connects now")
        match TcpStream.connect("127.0.0.1:${port}") {
            Ok(consume c) => c.close()
            Err(e) => println("c4 connect: ${e}")
        }
    }
    // Parked until the client above connects. Detached, so the scope does not wait for it.
    detach consume l {
        consume lis = l
        match lis.accept() {
            Ok(consume s) => s.close()
            Err(_) => ()
        }
        lis.close()
    }
    supervised {
        spawn {
            100.to_millis().sleep()
            println("c4 asking")
            ro got = ask(tx, 4)
            println("c4 reply ${got}")
            tx.close()
        }
        store(req.rx)
    }
    println("c4 done")
    with Os = real_os() { exit_process(0) }
}
