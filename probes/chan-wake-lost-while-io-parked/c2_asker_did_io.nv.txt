// c2: a1 (store in the body, asker in a spawn: GREEN), but the asker FIRST completes
// one socket round trip of its own (bind, connect, accept, write, read) and only then
// asks. When it asks, no fiber is parked in I/O.
module p1406.c2_asker_did_io

import std.concurrency.nvchan.{Chan, ChanTx, ChanRx}
import std.net.{TcpListener, TcpStream}
import std.os.{real_os, exit_process}
import std.time.duration

type Ask value {
    ro n int
    ro reply ChanTx[int]
}

fn store(rx ChanRx[Ask]) -> () {
    mut going = true
    while going {
        match rx.recv() {
            None => { going = false }
            Some(a) => { ro _ = a.reply.send(a.n * 10) }
        }
    }
}

fn ask(tx ChanTx[Ask], n int) -> int {
    ro pair = Chan[int].new(1)
    ro _ = tx.send({ n, reply: pair.tx })
    pair.rx.recv() ?? (0 - 1)
}

fn round_trip() Net Detach -> () {
    consume l = match TcpListener.bind("127.0.0.1:0") {
        Ok(consume x) => x
        Err(e) => panic("bind: ${e}")
    }
    ro port = l.local_port()
    supervised {
        spawn consume l {
            consume lis = l
            match lis.accept() {
                Ok(consume s) => {
                    ro _ = s.read_bytes(16)
                    ro _ = s.write_all("pong".bytes())
                    s.close()
                }
                Err(_) => ()
            }
            lis.close()
        }
        spawn {
            match TcpStream.connect("127.0.0.1:${port}") {
                Ok(consume c) => {
                    ro _ = c.write_all("ping".bytes())
                    ro t = c.read_text(16) ?? "<none>"
                    println("c2 round trip: ${t}")
                    c.close()
                }
                Err(e) => println("c2 connect: ${e}")
            }
        }
    }
}

fn main() Io Net Time Detach {
    detach {
        5000.to_millis().sleep()
        println("c2 WATCHDOG: still running after 5 s -- hung")
        with Os = real_os() { exit_process(3) }
    }
    ro req = Chan[Ask].new(8)
    ro tx = req.tx
    supervised {
        spawn {
            round_trip()
            println("c2 asking")
            ro got = ask(tx, 4)
            println("c2 reply ${got}")
            tx.close()
        }
        store(req.rx)
    }
    println("c2 done")
    with Os = real_os() { exit_process(0) }
}
