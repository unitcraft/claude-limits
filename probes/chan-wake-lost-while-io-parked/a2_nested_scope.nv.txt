// a2: as a1, but the asker is spawned inside a NESTED `supervised` that itself runs in
// a `spawn` -- the shape of a server: an accept loop in a fiber, each connection a
// fiber of the loop's own scope.
module p1406.a2_nested_scope

import std.concurrency.nvchan.{Chan, ChanTx, ChanRx}

type Ask value {
    ro n int
    ro reply ChanTx[int]
}

fn store(rx ChanRx[Ask]) -> () {
    mut going = true
    while going {
        match rx.recv() {
            None => { going = false }
            Some(a) => { ro _ = a.reply.send(a.n * 10) }
        }
    }
}

fn ask(tx ChanTx[Ask], n int) -> int {
    ro pair = Chan[int].new(1)
    ro _ = tx.send({ n, reply: pair.tx })
    pair.rx.recv() ?? (0 - 1)
}

fn connections(tx ChanTx[Ask]) Detach -> () {
    supervised {
        for i in 0..3 {
            spawn {
                ro got = ask(tx, i)
                println("a2 conn ${i} reply ${got}")
            }
        }
    }
}

fn main() Io Detach {
    ro req = Chan[Ask].new(8)
    ro tx = req.tx
    supervised {
        spawn {
            connections(tx)
            tx.close()
        }
        store(req.rx)
    }
    println("a2 done")
}
