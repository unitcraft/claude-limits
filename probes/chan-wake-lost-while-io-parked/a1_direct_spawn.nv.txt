// a1: the store answers in the body of `supervised`; the asker is a direct `spawn`.
module p1406.a1_direct_spawn

import std.concurrency.nvchan.{Chan, ChanTx, ChanRx}

type Ask value {
    ro n int
    ro reply ChanTx[int]
}

fn store(rx ChanRx[Ask]) -> () {
    mut going = true
    while going {
        match rx.recv() {
            None => { going = false }
            Some(a) => { ro _ = a.reply.send(a.n * 10) }
        }
    }
}

fn ask(tx ChanTx[Ask], n int) -> int {
    ro pair = Chan[int].new(1)
    ro _ = tx.send({ n, reply: pair.tx })
    pair.rx.recv() ?? (0 - 1)
}

fn main() Io Detach {
    ro req = Chan[Ask].new(8)
    ro tx = req.tx
    supervised {
        spawn {
            ro got = ask(tx, 4)
            println("a1 reply ${got}")
            tx.close()
        }
        store(req.rx)
    }
    println("a1 done")
}
